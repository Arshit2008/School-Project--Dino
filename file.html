import random
import tkinter as tk
from tkinter import messagebox
import mysql.connector
import pygame

# DATABASE CONFIGURATION
DB_host = "localhost"
DB_user = "root"
DB_password = "computer"
DB = input("Database Name: ").strip()

def connect():
    """Connect to selected MySQL database and return connection object."""
    try:
        return mysql.connector.connect(host=DB_host, user=DB_user, password=DB_password, database=DB)
    except Exception as err:
        print("DATABASE CONNECTION ERROR:", err)
        return None

def reset(db_name):
    """Delete selected database."""
    mycon = None
    try:
        mycon = mysql.connector.connect(host=DB_host, user=DB_user, password=DB_password)
        c = mycon.cursor()
        c.execute(f"DROP DATABASE IF EXISTS {db_name}")
    finally:
        if mycon: mycon.close()

def Db():
    """Initialize database and player, highscores, and game_history tables."""
    mycon = None
    try:
        mycon = mysql.connector.connect(host=DB_host, user=DB_user, password=DB_password)
        c = mycon.cursor()
        c.execute(f"CREATE DATABASE IF NOT EXISTS {DB}")
        c.execute(f"USE {DB}")
        c.execute("""CREATE TABLE IF NOT EXISTS player(
            players_id INT PRIMARY KEY AUTO_INCREMENT,
            username VARCHAR(20) UNIQUE NOT NULL
        )""")
        c.execute("""CREATE TABLE IF NOT EXISTS highscores(
            score_id INT PRIMARY KEY AUTO_INCREMENT,
            player_id INT UNIQUE,
            score INT NOT NULL,
            obstacles INT NOT NULL,
            FOREIGN KEY(player_id) REFERENCES player(players_id) ON DELETE CASCADE
        )""")
        c.execute("""CREATE TABLE IF NOT EXISTS game_history(
            history_id INT PRIMARY KEY AUTO_INCREMENT,
            player_id INT,
            score INT NOT NULL,
            obstacles INT NOT NULL,
            played_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY(player_id) REFERENCES player(players_id) ON DELETE CASCADE
        )""")
        mycon.commit()
    finally:
        if mycon: mycon.close()

def register_player(username):
    """Find or register a user handle and return primary key player_id."""
    mycon = connect()
    if not mycon: return None
    try:
        c = mycon.cursor()
        c.execute("SELECT players_id FROM player WHERE username = %s", (username,))
        check = c.fetchone()
        if check:
            return check[0]
        else:
            c.execute("INSERT INTO player (username) VALUES (%s)", (username,))
            mycon.commit()
            return c.lastrowid
    finally:
        mycon.close()

def save_history(player_id, score, obs):
    """Log individual match run into game_history."""
    mycon = connect()
    if mycon:
        c = mycon.cursor()
        c.execute("INSERT INTO game_history(player_id, score, obstacles) VALUES(%s, %s, %s)", (player_id, score, obs))
        mycon.commit()
        mycon.close()

def save_scores(player_id, score, obs):
    """Update highscore if new run surpasses previous personal best."""
    mycon = connect()
    if mycon:
        c = mycon.cursor()
        c.execute("SELECT score FROM highscores WHERE player_id = %s", (player_id,))
        res = c.fetchone()
        if res:
            if score > res[0]:
                c.execute("UPDATE highscores SET score = %s, obstacles = %s WHERE player_id = %s", (score, obs, player_id))
        else:
            c.execute("INSERT INTO highscores (player_id, score, obstacles) VALUES (%s, %s, %s)", (player_id, score, obs))
        mycon.commit()
        mycon.close()

#--- TKINTER GUI & STATS VIEWER ---
def show_stats():
    mycon = connect()
    if not mycon: return
    c = mycon.cursor()
    c.execute("SELECT username, score, obstacles FROM highscores h JOIN player p ON p.players_id = h.player_id ORDER BY score DESC")
    results = c.fetchall()
    mycon.close()
    stats = tk.Toplevel(root)
    stats.title("High-Scores")
    stats.geometry("480x320")
    tk.Label(stats, text="HIGH-SCORES", font=("Arial", 16, "bold")).pack(pady=10)
    for i, row in enumerate(results, start=1):
        tk.Label(stats, text=f"{i:<6} {row[0]:<15} Score: {row[1]:<6} Obs: {row[2]}", font=("Courier New", 10)).pack(anchor="w", padx=25)

def show_history():
    mycon = connect()
    if not mycon: return
    c = mycon.cursor()
    c.execute("SELECT p.username, g.score, g.obstacles, g.played_at FROM game_history g JOIN player p ON p.players_id = g.player_id ORDER BY g.history_id DESC LIMIT 20")
    history = c.fetchall()
    mycon.close()
    win = tk.Toplevel(root)
    win.title("Game History")
    win.geometry("500x320")
    tk.Label(win, text="Recent Games", font=("Arial", 16, "bold")).pack(pady=8)
    for i, row in enumerate(history, start=1):
        t_str = row[3].strftime("%m-%d %H:%M") if row[3] else ""
        tk.Label(win, text=f"{i:<3} {row[0]:<10} Score: {row[1]:<5} Obs: {row[2]:<3} [{t_str}]", font=("Courier New", 9)).pack(anchor="w", padx=15)

def game_over_window(player_id, final_score, obstacles):
    win = tk.Toplevel(root)
    win.title("Game Over")
    win.geometry("320x220")
    tk.Label(win, text="GAME OVER", font=("Arial", 16, "bold")).pack(pady=10)
    tk.Label(win, text=f"Final Score: {final_score} | Cleared: {obstacles}", font=("Arial", 11)).pack(pady=5)
    tk.Button(win, text="Retry", width=12, command=lambda: [win.destroy(), game_flow(player_id)]).pack(pady=4)
    tk.Button(win, text="Menu", width=12, command=lambda: [win.destroy(), root.deiconify()]).pack(pady=4)
    tk.Button(win, text="Exit", width=12, command=lambda: [win.destroy(), root.destroy()]).pack(pady=4)

def game_flow(player_id):
    score, obs = game(player_id)
    game_over_window(player_id, score, obs)

# PYGAME ENGINE
def game(player_id):
    pygame.init()
    width, height = 800, 400
    screen = pygame.display.set_mode((width, height))
    clock = pygame.time.Clock()
    run = True
    obs, dino_h, dino_w, dino_x, dino_y = 0, 50, 30, 100, 300
    dino_rect = pygame.Rect(dino_x, dino_y, dino_w, dino_h)
    cac_speed, ground_y, cactus_list = 6, 350, []
    spawn_timer, next_spawn = 0, 80
    velo, gravity, jump = 0, 1, -15
    font = pygame.font.Font(None, 36)

    while run:
        for event in pygame.event.get():
            if event.type == pygame.QUIT or (event.type == pygame.KEYDOWN and event.key == pygame.K_ESCAPE):
                run = False
            if event.type == pygame.KEYDOWN and event.key == pygame.K_SPACE and dino_y >= ground_y - dino_h:
                velo = jump

        velo += gravity
        dino_y += velo
        if dino_y >= ground_y - dino_h:
            dino_y, velo = ground_y - dino_h, 0
        dino_rect.y = dino_y

        screen.fill((255, 255, 255))
        screen.blit(font.render(f"Score: {obs * 5} Speed: {cac_speed}", True, (0, 0, 0)), (10, 10))
        pygame.draw.line(screen, (0, 0, 0), (0, ground_y), (width, ground_y), 3)
        pygame.draw.rect(screen, (0, 0, 0), dino_rect)

        spawn_timer += 1
        if spawn_timer >= next_spawn:
            ch, cw = random.choice([(25, 15), (60, 25), (40, 45)])
            cactus_list.append([pygame.Rect(width, ground_y - ch, cw, ch), False])
            spawn_timer, next_spawn = 0, random.randint(max(28, int(450 / cac_speed)), max(45, int(750 / cac_speed)))

        for c in cactus_list:
            c[0].x -= cac_speed
            pygame.draw.rect(screen, (0, 150, 0), c[0])
            if c[0].right < dino_x and not c[1]:
                obs += 1
                c[1] = True
                if obs % 5 == 0 and cac_speed < 30:
                    cac_speed += 2

            if dino_rect.colliderect(c[0]):
                save_history(player_id, obs * 5, obs)
                save_scores(player_id, obs * 5, obs)
                run = False
                break

        cactus_list = [c for c in cactus_list if c[0].right >= 0]
        pygame.display.update()
        clock.tick(60)

    pygame.quit()
    return obs * 5, obs

EXPLANATION FOR MAJOR PARTS (PART 1)

Database Connectivity, Schema Relational Integrity & Data Logic

Database Init

connect() & Db()
  • Connection Handshake & DDL Automation: connect() initializes the mysql.connector session with error-trapping to prevent abrupt crashes if the MySQL server is offline.
  • Db() automatically runs DDL commands (CREATE DATABASE IF NOT EXISTS) to set up normalized tables without manual SQL intervention.

Schema Design

Relational Integrity
  • Primary Keys & Cascade Architecture:
  • player: Primary table storing unique player handles with auto-incrementing ID.
  • highscores: 1-to-1 relationship with unique player_id foreign key; maintains the highest points scored.
  • game_history: 1-to-Many relationship logging every match with automated TIMESTAMP. Uses ON DELETE CASCADE for clean referential integrity.

Player Auth

register_player()
  • Sanitization & Identifier Fetching: Queries for an existing username using parameterized placeholders (%s) to prevent SQL injection vulnerabilities.
  • If found, it returns the existing players_id. If not, it executes an INSERT and captures cursor.lastrowid.

Score Tracking

save_scores() / history
  • Highscore Update vs. Run Auditing: save_scores() compares the current run against previous personal best, performing an UPDATE only on high-score breakthroughs.
  • save_history() continuously appends complete run details (score, cleared obstacles, timestamp) for statistical auditing.

EXPLANATION FOR MAJOR PARTS (PART 2)

Tkinter GUI Dashboard & Pygame 2D Physics Mechanics

GUI Dashboard

Tkinter Windows
  • Menu Navigation & Statistical Queries: Main Window validates user input before launching the game window.
  • show_stats(): Executes an SQL JOIN between player and highscores with ORDER BY score DESC to render a ranked leaderboard.
  • show_history(): Fetches recent match logs (LIMIT 20) with formatted timestamps.
  • game_over_window(): Modal dialog allowing direct retry, menu navigation, or clean process termination.

Jump Physics

Discrete Mechanics
  • Discrete Velocity & Gravity Modeling: Pressing SPACE applies an initial upward impulse: velo = -15.
  • On each frame, gravity increments velocity: velo += gravity and dino_y += velo.
  • Ground boundary detection halts vertical displacement when dino_y >= ground_y - dino_h.

Dynamic Obstacles

Procedural Spawning
  • Procedural Generation & Balancing: Randomly selects from 3 distinct cactus dimensions (small, tall, wide).
  • Dynamically scales spawn intervals inversely with movement speed so obstacles never overlap or create impossible jumps.
  • Increases game speed by +2 every 5 successfully cleared cacti.

Collision & Loop

Pygame 60 FPS
  • Real-Time Detection & Memory Management: Performs bounding-box collision detection using pygame.Rect.colliderect() at 60 FPS.
  • Offscreen obstacles (c[0].right < 0) are pruned via list filtering to prevent memory leaks.
  • Triggers automated database save functions immediately upon collision before destroying the Pygame window.
